# Tests de Wilson

Tests de bout en bout dans Chromium (Playwright), avec faux micro, fausse caméra et Gemini simulé
(aucune clé ni requête réelle : la clé utilisée est un exemple factice).

```sh
pip install playwright && playwright install chromium   # une fois
./make_fixtures.sh                                       # fausses vidéos de caméra (.y4m) à partir des photos
python3 make_voix.py                                     # voix de synthèse pour l'oreille du mode voiture (espeak-ng + mbrola-fr1/fr4)
cd .. && python3 -m http.server 8765 --bind 127.0.0.1 &  # servir l'appli (dossier racine du dépôt)
cd tests && for t in e2e.py e2e_eco.py e2e_echo.py e2e_conn.py e2e_key.py e2e_update.py stream_unit.py e2e_gaze.py e2e_speed.py e2e_remote.py e2e_sound.py e2e_live_search.py e2e_attention.py e2e_sync.py e2e_vision_health.py e2e_crm.py e2e_car.py e2e_journal_2510.py e2e_route.py e2e_question.py e2e_maison.py e2e_adresse.py e2e_outils.py e2e_regard_sur.py e2e_bluetooth.py e2e_doublon.py e2e_voiture_0411.py e2e_oreille_0411.py e2e_crm_0411.py e2e_sourd_0411.py e2e_famille.py e2e_memoire.py e2e_journal_0511.py e2e_journal_0511b.py e2e_journal_0511c.py e2e_journal_0511d.py e2e_android.py e2e_android_micro.py e2e_claude.py e2e_journal_1011.py e2e_journal_1211.py; do python3 $t; done
node voiture_unit.mjs                                    # règles du protocole voiture sur les phrases du journal
node journal_1211_unit.mjs                               # journal du 12/11/26 : GPS, heure, écriture, FMA ≠ Bouygues, contact du compte rendu
python3 e2e_famille_modeles.py                           # visages et sons avec les vrais modèles (lent)
node apps_script_unit.js                                 # script Apps Script du journal, avec une fausse feuille
node oreille_unit.mjs                                    # oreille du mode voiture sur les voix de synthèse
node maison_unit.mjs                                     # module maison (faux Home Assistant)
node relais_unit.mjs                                     # relais Claude du PC et CRM en lecture seule (PLAN 10)
```
